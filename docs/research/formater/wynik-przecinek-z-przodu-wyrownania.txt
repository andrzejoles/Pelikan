### 20-kursor
DECLARE
  CURSOR c_emp(p_dept NUMBER
             , p_job VARCHAR2 DEFAULT 'CLERK') IS SELECT empno
                                                       , ename
                                                       , sal
                                                  FROM   emp
                                                  WHERE  deptno = p_dept
                                                  AND    job = p_job
                                                  ORDER BY sal DESC
                                                  FOR UPDATE OF sal NOWAIT;
  r c_emp%ROWTYPE;
BEGIN
  OPEN c_emp(10
           , 'ANALYST');
  LOOP
    FETCH c_emp INTO r;
    EXIT WHEN c_emp%notfound;
    UPDATE emp
    SET    sal = r.sal * 1.05
    WHERE  CURRENT OF c_emp;
  END LOOP;
  CLOSE c_emp;
END;

### 21-forall-bulk
DECLARE
  TYPE t_emp IS TABLE OF emp%ROWTYPE;
  l_emp t_emp;
BEGIN
  SELECT *
  BULK COLLECT INTO l_emp
  FROM   emp
  WHERE  deptno = 10;
  FORALL i IN 1..l_emp.count SAVE EXCEPTIONS INSERT INTO emp_copy VALUES l_emp(i);
  FORALL i IN INDICES OF l_emp UPDATE emp SET sal = l_emp(i).sal WHERE empno = l_emp(i).empno;
EXCEPTION
  WHEN OTHERS THEN
    IF SQLCODE = -24381 THEN
    FOR j IN 1..sql%bulk_exceptions.count LOOP
      dbms_output.put_line(sql%bulk_exceptions(j).error_index || ': ' || SQLERRM(-sql%bulk_exceptions(j).error_code));
    END LOOP;
  ELSE
    RAISE;
  END IF;
END;

CREATE TABLE order_items(order_id NUMBER(10) NOT NULL
                       , line_no NUMBER(5) NOT NULL
                       , product_id NUMBER(10) NOT NULL
                       , qty NUMBER(10, 3) DEFAULT 1 NOT NULL
                       , price NUMBER(14, 2)
                       , discount_pct NUMBER(5, 2) DEFAULT 0 CONSTRAINT oi_disc_ck CHECK (discount_pct BETWEEN 0 AND 100)
                       , note VARCHAR2(4000 CHAR)
                       , created_at TIMESTAMP(6) WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL
                       , CONSTRAINT order_items_pk PRIMARY KEY (order_id, line_no) USING INDEX TABLESPACE idx_ts
                       , CONSTRAINT oi_order_fk FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE
                       , CONSTRAINT oi_prod_fk FOREIGN KEY (product_id) REFERENCES products(product_id) DEFERRABLE INITIALLY DEFERRED
                       , CONSTRAINT oi_qty_ck CHECK (qty > 0 AND (price IS NULL OR price >= 0))) TABLESPACE data_ts PCTFREE 10 STORAGE(INITIAL 1 m NEXT 1 m) PARTITION BY RANGE (order_id) INTERVAL(1000000) (PARTITION p0 VALUES less than(1000000));

### 56-trigger-compound
CREATE OR REPLACE TRIGGER oi_compound FOR INSERT OR UPDATE ON order_items
COMPOUND TRIGGER TYPE t_ids IS TABLE OF NUMBER INDEX BY PLS_INTEGER;
g_ids t_ids;
BEFORE statement IS
BEGIN
  g_ids.DELETE;
END BEFORE statement;
AFTER EACH ROW IS
BEGIN
  g_ids(g_ids.count + 1) := :new.order_id;
END AFTER EACH ROW;
AFTER statement IS
BEGIN
  FORALL i IN 1..g_ids.count UPDATE orders SET modified = SYSDATE WHERE order_id = g_ids(i);
END AFTER statement;
END oi_compound;

### 60-type-body
CREATE OR REPLACE TYPE BODY t_money AS
  CONSTRUCTOR FUNCTION t_money(p_amount NUMBER) RETURN SELF AS
    result IS
  BEGIN
    SELF.amount := p_amount;
    SELF.currency := 'PLN';
    RETURN;
  END;
  MEMBER FUNCTION to_string RETURN VARCHAR2 IS
  BEGIN
    RETURN to_char(amount
                 , 'FM999G990D00') || ' ' || currency;
  END;
  MAP MEMBER FUNCTION sort_key RETURN NUMBER IS
  BEGIN
    RETURN amount;
  END;
  STATIC FUNCTION zero RETURN t_money IS
  BEGIN
    RETURN t_money(0);
  END;
END;

### 62-pakiet-body-pipelined
CREATE OR REPLACE PACKAGE BODY pkg_items AS
  CURSOR c_items(p_order_id NUMBER) RETURN order_items%ROWTYPE IS SELECT *
                                                                  FROM   order_items
                                                                  WHERE  order_id = p_order_id;
  FUNCTION items(p_order_id NUMBER) RETURN t_money_tab PIPELINED IS
  BEGIN
    FOR r IN c_items(p_order_id) LOOP
      PIPE ROW (t_money(r.qty * r.price));
    END LOOP;
    RETURN;
  END items;
  FUNCTION total(p_order_id NUMBER) RETURN NUMBER RESULT_CACHE DETERMINISTIC PARALLEL_ENABLE IS
    v NUMBER;
  BEGIN
    SELECT sum(qty * price)
    INTO   v
    FROM   order_items
    WHERE  order_id = p_order_id;
    RETURN v;
  END total;
  PROCEDURE add_item(p_order_id   IN  NUMBER
                   , p_product_id IN  NUMBER
                   , p_qty        IN  NUMBER DEFAULT 1
                   , p_price      IN  NUMBER DEFAULT NULL
                   , p_line_no    OUT NUMBER) IS
  BEGIN
    INSERT INTO order_items(order_id
                          , line_no
                          , product_id
                          , qty
                          , price)
    VALUES (p_order_id
          , (SELECT nvl(max(line_no), 0) + 1
             FROM   order_items
             WHERE  order_id = p_order_id)
          , p_product_id
          , p_qty
          , p_price)
    RETURNING line_no
    INTO   p_line_no;
  END add_item;
END pkg_items;
